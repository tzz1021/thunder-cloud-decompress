def _alist_list(path: str) -> list:
    """通用 Alist API 调用：列出 path 下的内容。返回 content 列表或空列表。"""
    import requests as _req
    try:
        resp = _req.post(
            f"{ALIST_BASE}/api/fs/list",
            headers={"Authorization": ALIST_TOKEN},
            json={"path": path, "password": "", "page": 1, "per_page": 0, "refresh": True},
            timeout=10
        )
        data = resp.json()
        if data.get("code") != 200:
            return []
        content = data.get("data", {})
        if content is None:
            return []
        content = content.get("content")
        return content if content else []
    except Exception as e:
        logger.warning(f"Alist API 请求失败 ({path}): {e}")
        return []


def check_folder_has_files(folder_name: str) -> bool:
    """

    路径: /online_prase_site/文件夹名

    返回 False: 文件夹不存在或为空

    # Alist API 处理路径时直接支持中文/全角字符，不需要 URL 编码

    return False

## 复盘：三个高频报错点

### 1️⃣ 点击"添加链接"（`click_add_link`） → `element not interactable`

```python
# browser_engine.py L359
def click_add_link(self):
    add_link = WebDriverWait(self.driver, 5).until(
        EC.presence_of_element_located((By.XPATH, "//span[text()='添加链接']/.."))
    )
    self.driver.execute_script("arguments[0].click();", add_link)
    time.sleep(1.5)
```

❌ **问题：** `presence_of_element_located` 只检查 DOM 存在，不检查是否可见/可点击。PlusMenu 弹出动画还没完成就点击了，菜单项被遮挡 → `element not interactable`。且**无容错**，抛异常直接到 `main_loop` 的 except 块标记 failed。

✅ **建议加：** 等待可点击 + JS 点击降级

```python
def click_add_link(self):
    try:
        add_link = WebDriverWait(self.driver, 5).until(
            EC.element_to_be_clickable((By.XPATH, "//span[text()='添加链接']/.."))
        )
        add_link.click()
    except:
        # 降级：presence + JS click
        add_link = self.driver.find_element(By.XPATH, "//span[text()='添加链接']/..")
        self.driver.execute_script("arguments[0].click();", add_link)
    time.sleep(1.5)
```

❌ **问题：** 一次性搜索所有 `td-tree-node`，如果树形目录没加载完或者 dialog 没渲染，`all_nodes` 为空循环直接跳过 → 返回 `None`。**返回 None 后 `step2` 直接标记 failed**，没有重试。

✅ **`step2_cloud_upload.py` 已有 3 次重试**（`for attempt in range(1, 4)`），但问题在**重试之间没有重新搜索节点**——3 次重试都在第一次 `find_tree_node_by_label` 返回的同一个 `target_node` 上操作。正确做法是：**每轮重试都重新 `find_tree_node_by_label`**。

### 3️⃣ 时间戳文件夹无法被点击（`click_tree_node_select`）

```python
# browser_engine.py L516
def click_tree_node_select(self, node):
    content = node.find_element(By.XPATH, "./div[contains(@class,'td-tree-node__content')]")
    candidates = []
    candidates.append(content.find_element(By.XPATH, "./span[2]"))  # 图标
```

❌ **问题：** 三选一偶尔点到被遮挡的元素（比如复选框被展开箭头遮住）。且**无 JS click 降级**——原生的 `click()` 被遮挡就抛 `element not interactable`。
