async def process_file(req: ProcessRequest):
    """
    主处理入口：
    1. 如果是直链 → 云添加下载

    3. 检查文件大小(≥1GB)

                share_id=req.share_id,
                file_ids=file_ids,
                pass_code_token=req.pass_code_token,

@app.get("/decompress-sse/{task_id}")
async def decompress_sse(task_id: str, interval: float = Query(2.0)):
    """
    SSE 实时推送解压进度

    前端用法:
      const evtSource = new EventSource('/decompress-sse/TASK_ID');
      evtSource.onmessage = (e) => {
        const data = JSON.parse(e.data);
        console.log(data.progress + '%');
      };
    """
    async def event_generator():
        client = get_client()
        while True:
            try:
                result = client.get_decompress_progress(task_id=task_id)

                # 判断是否完成
                status = result.get("status", "unknown")
                progress = result.get("progress", 0)

                yield f"data: {json.dumps(result)}\n\n"

                if status in ("completed", "finish", "done"):  # 具体值待确认
                    yield f"data: {json.dumps({'status': 'completed', 'progress': 100})}\n\n"
                    break

                await asyncio.sleep(interval)

            except XunleiAPIError as e:
                yield f"data: {json.dumps({'error': str(e)})}\n\n"
                break

    return StreamingResponse(
        event_generator(),
        media_type="text/event-stream",
        headers={
            "Cache-Control": "no-cache",

        },
