2. `goog:loggingPrefs + get_log("performance")` — Edge 不支持 `performance` log type
3. CDP `Network.requestWillBeSent` — Selenium 没有该事件命令
4. 让 token_keeper 同时抓 device-id 和 client-id — 但 JS 拦截到的 device-id 也跟实际的绑定的不一样

def run(engine, dict2: dict, processed_set: set,
        new_folder_name: str = "",
        max_wait_seconds: int = 300) -> dict:
    """
    监听文件夹修改时间变化，完成下载后重命名+解压。
    
    new_folder_name: 刚创建的时间戳文件夹名，用于快速检查秒传。
                     如果文件已在，直接走重命名→解压，不再监听。
    
    返回:
      {
        "folder_name": str,       # 处理完成的文件夹名
        "success": bool,
        "message": str
      }
    """
    elapsed = 0
    result = {"folder_name": "", "success": False, "message": ""}

    # ── [快速方式] 进入时间戳文件夹 → 检查 pan-list ──
    if new_folder_name:
        import random as _r

        logger.info(f"快速方式: 检查文件夹 {new_folder_name} 内文件是否存在")

        # step2 结束时已在 target_dir，但 URL 跳转可能无法刷新 SPA
        # 先回到根目录，再点击进入 target_dir 确保渲染完整
        engine.navigate_to(PAN_BASE)
        time.sleep(_r.uniform(1.0, 2.0))
        engine.navigate_to_target_dir()
        time.sleep(_r.uniform(2.0, 3.0))

        # 点击刚刚创建的时间戳文件夹进入
        engine.click_file_by_title(new_folder_name)
        time.sleep(_r.uniform(1.0, 3.0))

        try:
            pan_list = engine.driver.find_element(By_CSS_SELECTOR, PAN_LIST_CSS)
            if pan_list:
                logger.info(f"快速方式: 文件夹 {new_folder_name} 内已有文件，直接处理")
                result = _rename_and_decompress(engine, new_folder_name)
                processed_set.add(new_folder_name)
                # 更新 dict2
                current_times = scan_folder_times(engine)
                if new_folder_name in current_times:
                    dict2[new_folder_name] = current_times[new_folder_name]
                result["folder_name"] = new_folder_name
                return result
        except Exception:
            pass

        # 空值 → 回到 target_dir，走传统方式
        logger.info(f"快速方式: 文件夹 {new_folder_name} 内无文件，切换到传统监听")
        engine.navigate_to(PAN_BASE)
        time.sleep(1)
        engine.navigate_to_target_dir()
    else:
        # 没传 new_folder_name，直接走传统方式
        engine.navigate_to(PAN_BASE)
        time.sleep(2)
        engine.navigate_to_target_dir()

    # ── [传统方式] 扫描所有文件夹，对比修改时间 ──
    logger.info("开始监听文件夹修改时间变化")

    while elapsed < max_wait_seconds:
        current_times = scan_folder_times(engine)

        for folder_name, current_date in current_times.items():
            # 跳过已处理的文件夹
            if folder_name in processed_set:
                continue

            if folder_name in dict2:
                prev_date = dict2[folder_name]
                if current_date != prev_date:
                    logger.info(f"检测到修改时间变化: {folder_name}  {prev_date} → {current_date}")
                    child_url = CHILD_DIR_TEMPLATE.format(
                        foldername=urllib.parse.quote(folder_name, safe=''))
                    engine.navigate_to(child_url)
                    time.sleep(2)

                    process_result = _rename_and_decompress(engine, folder_name)
                    processed_set.add(folder_name)
                    dict2[folder_name] = current_date
                    process_result["folder_name"] = folder_name
                    return process_result

            else:
                # 首次出现，记录初始时间
                dict2[folder_name] = current_date
                logger.info(f"记录初始时间: {folder_name} = {current_date}")

        # 更新 dict2 中的值
        for folder_name, current_date in current_times.items():
            if folder_name in dict2:
                dict2[folder_name] = current_date

        time.sleep(3)
        elapsed += 3

        if elapsed % 30 == 0:
            logger.info(f"监听中... 已等待 {elapsed}s, 跟踪 {len(dict2)} 个文件夹")

    result["message"] = f"超时: {max_wait_seconds} 秒内未检测到修改时间变化"
    logger.warning(result["message"])
    return result
