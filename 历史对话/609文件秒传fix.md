  4. 右键压缩包 → 重命名为 文件夹名.zip
  5. 确认重命名
  6. 单击压缩包弹出解压窗口
  7. 点击"解压全部文件"
  8. 标记该文件夹已处理

调用者需传入：
  - engine: BrowserEngine 实例
  - dict1: 文件夹名 → data-file-id 映射（来自 step1）
  - dict2: 文件夹名 → 修改时间 映射（本次扫描用）
  - processed_set: 已处理的文件夹名集合
  - max_wait_seconds: 单次最大等待时间（默认 300 秒 = 5 分钟）

import logging
import urllib.parse
from pathlib import Path

logger = logging.getLogger("step3_monitor")

TARGET_DIR = "https://pan.xunlei.com/?path=%2F%E5%9C%A8%E7%BA%BF%E8%A7%A3%E5%8E%8B%E7%AB%99%E7%82%B9%E9%A3%9F%E7%94%A8"

FILE_ITEM_XPATH = "/html/body/div[1]/div/div/div/div[2]/div[2]/section[2]/ul/li[1]/div/div[2]/div[1]"


def scan_folder_times(engine) -> dict:
    """
    扫描所有文件夹的修改时间。
    返回 dict: {文件夹名: 修改时间字符串}
    
    选择器：
      - folder_list_item: /html/body/.../ul/li[{index}]
      - folder_name_span: ./div/div[2]/div[1]/div[2]/a/span
      - folder_date_div:  ./div/div[2]/div[2]/div[3]
    """
    result = {}
    folder_list = engine.get_folder_info_list()
    for f in folder_list:
        result[f["name"]] = f["date"]
    return result


def run(engine, dict2: dict, processed_set: set,
        max_wait_seconds: int = 300) -> dict:
    """
    监听文件夹修改时间变化，完成下载后重命名+解压。

                    time.sleep(2)

                    new_name = f"{folder_name}.zip"

                    # 右键调用
                    engine.right_click_by_title(original_title)
                    # 点击"重命名"
                    engine.click_context_menu_item("重命名")
                    time.sleep(1)

                    # 输入新名称
                    input_el = engine.find_rename_input()
                    input_el.clear()
                    input_el.send_keys(new_name)
                    time.sleep(0.5)

                    # 确认重命名
                    engine.click_rename_confirm()
                    logger.info(f"已重命名: {original_title} → {new_name}")

                    # ── 步骤4: 单击压缩包弹出解压窗口 ──
                    engine.click_file_by_title(new_name)

                    processed_set.add(folder_name)

                    dict2[folder_name] = current_date

                    result["success"] = True
